using System;
using System.IO;
using System.Linq;
using Unity.AI.Navigation;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.AI;

namespace ROM.EditorTools
{
    public static partial class CiudadDelSolNaveJardinesPreview
    {
        public static void ProbeGardenInsetInBatch()
        {
            const string expected = "4192D89D77FA3BA97201D39124FE0B574AB2BF49031738A534C3BC241774F2BD";
            const string output = "Backups/CiudadDelSol/JardinesRaid/MargenInterior";
            const float margin = .35f;
            var court = new Bounds(new Vector3(-119.5f, 4.8f, -49.5f), new Vector3(20, 2.5f, 16));
            var scene = EditorSceneManager.OpenScene(Original, OpenSceneMode.Single);
            Require(Hash() == expected && !scene.isDirty && NavMesh.CalculateTriangulation().indices.Length == 0,
                "La escena de origen o la navegación inicial cambiaron.");
            var obj = new GameObject("Prueba de margen · solo en memoria");
            try
            {
                obj.transform.position = court.center;
                var surface = obj.AddComponent<NavMeshSurface>();
                surface.agentTypeID = 0;
                surface.collectObjects = CollectObjects.Volume;
                surface.useGeometry = NavMeshCollectGeometry.PhysicsColliders;
                surface.center = Vector3.zero;
                surface.size = court.size - new Vector3(2 * margin, 0, 2 * margin);
                surface.BuildNavMesh();
                var mesh = NavMesh.CalculateTriangulation();
                float area = 0;
                for (int i = 0; i < mesh.indices.Length; i += 3)
                {
                    var a = mesh.vertices[mesh.indices[i]];
                    var b = mesh.vertices[mesh.indices[i + 1]] - a;
                    var c = mesh.vertices[mesh.indices[i + 2]] - a;
                    area += Mathf.Abs(Vector3.Cross(b, c).y) * .5f;
                }
                var city = scene.GetRootGameObjects().Single(g => g.name == "Dungeon_Ciudad_del_Sol_Generated");
                var routes = city.GetComponentsInChildren<Transform>().Where(t => t.name.Contains("ancho util")).ToArray();
                int conflicts = routes.SelectMany(t => t.GetComponentsInChildren<Collider>())
                    .Count(c => c.enabled && !c.isTrigger && c.bounds.Intersects(court));
                int links = scene.GetRootGameObjects().Sum(r => r.GetComponentsInChildren<NavMeshLink>(true).Length);
                var points = new[] { new Vector3(-122.3f, 4, -52), new Vector3(-115.7f, 4, -46.5f),
                    new Vector3(-124, 4, -47), new Vector3(-115, 4, -47) };
                var filter = new NavMeshQueryFilter { agentTypeID = 0, areaMask = NavMesh.AllAreas };
                var projected = points.Select(p => NavMesh.SamplePosition(p, out var hit, .85f, filter) &&
                    court.Contains(hit.position) && Mathf.Abs(hit.position.y - 4) < .2f ? (Vector3?)hit.position : null).ToArray();
                int complete = 0;
                foreach (var a in projected.Where(p => p.HasValue))
                foreach (var b in projected.Where(p => p.HasValue && p != a))
                {
                    var path = new NavMeshPath();
                    if (NavMesh.CalculatePath(a.Value, b.Value, filter, path) &&
                        path.status == NavMeshPathStatus.PathComplete && path.corners.All(court.Contains)) complete++;
                }
                int outside = mesh.vertices.Count(v => !court.Contains(v));
                int otherHeight = mesh.vertices.Count(v => Mathf.Abs(v.y - 4) >= .2f);
                string bounds = mesh.vertices.Length == 0 ? "Sin vértices" :
                    $"X={mesh.vertices.Min(v=>v.x):F3}…{mesh.vertices.Max(v=>v.x):F3}; " +
                    $"Y={mesh.vertices.Min(v=>v.y):F3}…{mesh.vertices.Max(v=>v.y):F3}; Z={mesh.vertices.Min(v=>v.z):F3}…{mesh.vertices.Max(v=>v.z):F3}";
                var report = new[] {
                    $"Patio 20 × 16 m. Margen X/Z por lado {margin:F2} m. Volumen de bake {surface.size.x:F2} × {surface.size.z:F2} m.",
                    $"Área navegable proyectada {area:F2} m² / 320 m² ({area/320:P1}); {mesh.indices.Length/3} triángulos, {mesh.vertices.Length} vértices.",
                    $"{outside} vértices fuera del patio y {otherHeight} fuera de Y=4±0,2. {bounds}.",
                    $"Referencias {projected.Count(p=>p.HasValue)}/4; recorridos dirigidos {complete}/12.",
                    $"Rutas inspeccionadas {routes.Length}/9; colliders de ruta que cruzan el patio {conflicts}; enlaces NavMesh {links}.",
                    "Referencias proyectadas: " + string.Join("; ", projected.Select(p=>p.HasValue ? p.Value.ToString("F3") : "sin proyección")),
                    $"Escena SHA-256: {Hash()}"
                };
                File.WriteAllLines(Path.Combine(output, "Resultado.txt"), report);
                Require(outside == 0 && otherHeight == 0 && complete == 12 && conflicts == 0 &&
                    routes.Length == 9 && links == 0 && area > 0 && Hash() == expected, "El margen no superó el control estricto.");
                Debug.Log("JARDINES_MARGEN_OK");
            }
            finally
            {
                UnityEngine.Object.DestroyImmediate(obj);
                Require(Hash() == expected, "Cambió el archivo original.");
            }
        }
    }
}
